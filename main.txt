function GetCurrentCharacter()
    local ok, res = pcall(function() return LocalPlayer.Data.Character.Value end)
    if ok and res then return res end
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    return hum and hum:GetAttribute("CharacterName") or "Gon"
end

function HasAbility4(characterName)
    local ok, res = pcall(function()
        local chars = ReplicatedStorage:WaitForChild("Characters")
        local folder = chars:FindFirstChild(characterName)
        local ab = folder and folder:FindFirstChild("Abilities")
        return ab and ab:FindFirstChild("4") ~= nil
    end)
    return ok and res
end

function FindNearestPlayer()
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    local nearest, dist = nil, math.huge
    for _,p in pairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            local tr = p.Character:FindFirstChild("HumanoidRootPart")
            local th = p.Character:FindFirstChild("Humanoid")
            if tr and th then
                local hp = th:GetAttribute("Health") or th.Health
                if hp and hp > 0 then
                    local d = (hrp.Position - tr.Position).Magnitude
                    if d < dist then dist = d; nearest = p end
                end
            end
        end
    end
    return nearest
end

function FindNearestPlayerAura()
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    local nearest, dist = nil, math.huge
    for _,p in pairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            local tr = p.Character:FindFirstChild("HumanoidRootPart")
            local th = p.Character:FindFirstChild("Humanoid")
            if tr and th then
                local hp = th:GetAttribute("Health") or th.Health
                if hp and hp > 0 then
                    local d = (hrp.Position - tr.Position).Magnitude
                    if d < dist and d <= 150 then dist = d; nearest = p end
                end
            end
        end
    end
    return nearest
end

local function GetCFrameString(cf)
    local x, y, z, r00, r01, r02, r10, r11, r12, r20, r21, r22 = cf:GetComponents()
    return string.format("%f,%f,%f,%f,%f,%f,%f,%f,%f,%f,%f,%f", x, y, z, r00, r01, r02, r10, r11, r12, r20, r21, r22)
end

function _G.LagServer:UseAbility4()
    local charName = GetCurrentCharacter()
    if not HasAbility4(charName) then return end
    local target = FindNearestPlayer()
    if not target then return end
    local targetChar = target.Character
    local targetCF = targetChar and targetChar:FindFirstChild("HumanoidRootPart") and targetChar.HumanoidRootPart.CFrame or CFrame.new()

    pcall(function()
        local ability = ReplicatedStorage.Characters[charName].Abilities["4"]
        ReplicatedStorage.Remotes.Abilities.Ability:FireServer(ability, 9000000)
        local actions = {377,380,383,384,385,387,389}
        for i=1,7 do
            local args = {
                ability, charName..":Abilities:4", i, 9000000,
                { HitboxCFrames = {targetCF,targetCF}, BestHitCharacter = targetChar, HitCharacters = {targetChar}, Ignore = i>2 and {ActionNumber1={targetChar}} or {}, DeathInfo = {}, BlockedCharacters = {}, HitInfo = { IsFacing = not (i==1 or i==2), IsInFront = i<=2, Blocked = i>2 and false or nil }, ServerTime = tick(), Actions = i>2 and {ActionNumber1={}} or {}, FromCFrame = targetCF },
                "Action"..actions[i], i==2 and 0.1 or nil
            }
            if i==7 then
                args[5].RockCFrame = targetCF
                args[5].Actions = { ActionNumber1 = { [target.Name] = { StartCFrameStr = tostring(targetCF.X)..","..tostring(targetCF.Y)..","..tostring(targetCF.Z)..",0,0,0,0,0,0,0,0,0", ImpulseVelocity = Vector3.new(1901,-25000,291), AbilityName = "4", RotVelocityStr = "0,0,0", VelocityStr = "1.900635,0.010867,0.291061", Duration = 2, RotImpulseVelocity = Vector3.new(5868,-6649,-7414), Seed = math.random(1,1e6), LookVectorStr = "0.988493,0,0.151268" } } }
            end
            ReplicatedStorage.Remotes.Combat.Action:FireServer(unpack(args))
        end
    end)
end

function _G.LagServer:Start()
    if self.loopThread then return end
    self.running = true
    self.loopThread = task.spawn(function()
        while self.running do
            if FindNearestPlayer() then
                self:UseAbility4()
                pcall(function()
                    local c = GetCurrentCharacter()
                    ReplicatedStorage.Remotes.Abilities.AbilityCanceled:FireServer(ReplicatedStorage.Characters[c].Abilities["4"])
                end)
                task.wait(0.01)
            else task.wait(0.5) end
        end
        self.loopThread = nil
    end)
end

-- ==========================================
function _G.WallCombo:UseCombo()
    local target = FindNearestPlayer()
    if not target then return end
    local myHRP = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    local tChar = target.Character
    local tHRP = tChar and tChar:FindFirstChild("HumanoidRootPart")
    if not myHRP or not tHRP then return end
    if (myHRP.Position - tHRP.Position).Magnitude > 50 then return end

    local charName = GetCurrentCharacter()
    local wallCombo = ReplicatedStorage.Characters[charName] and ReplicatedStorage.Characters[charName]:FindFirstChild("WallCombo")
    if not wallCombo then return end

    local tCF, myCF = tHRP.CFrame, myHRP.CFrame
    local token = math.random(100000, 999999)
    local Ability = ReplicatedStorage.Remotes.Abilities.Ability
    local Action  = ReplicatedStorage.Remotes.Combat.Action
    local Cancel  = ReplicatedStorage.Remotes.Abilities.AbilityCanceled
    local EmoteClaim = ReplicatedStorage.Remotes.Combat:FindFirstChild("EmoteClaim")

    if EmoteClaim then pcall(function() EmoteClaim:FireServer() end) end
    pcall(function() ReplicatedStorage.Remotes.Combat.Melee:FireServer(tChar) end)
    pcall(function() ReplicatedStorage.Remotes.Combat.RagdollHit:FireServer(tChar) end)
    pcall(function() ReplicatedStorage.Remotes.Effects.VFX:FireServer("Hit", tHRP.Position) end)
    pcall(function() Ability:FireServer(wallCombo, token, nil, tChar, tHRP.Position) end)

    pcall(function() Action:FireServer(wallCombo, "Characters:"..charName..":WallCombo", 1, token, { HitboxCFrames = {}, BestHitCharacter = tChar, HitCharacters = {tChar}, Ignore = {}, DeathInfo = {}, Actions = {}, HitInfo = { Blocked = false, IsFacing = true, IsInFront = true, GetUp = true }, BlockedCharacters = {}, ServerTime = tick(), FromCFrame = myCF }, "Action69", 0) end)
    pcall(function() Action:FireServer(wallCombo, "Characters:"..charName..":WallCombo", 2, token, { HitboxCFrames = {tCF}, BestHitCharacter = tChar, HitCharacters = {tChar}, Ignore = { ActionNumber1 = {tChar} }, DeathInfo = {}, Actions = { ActionNumber1 = {} }, HitInfo = { Blocked = false, IsFacing = true, IsInFront = true }, BlockedCharacters = {}, ServerTime = tick(), FromCFrame = myCF }, "Action70") end)
    pcall(function() Action:FireServer(wallCombo, "Characters:"..charName..":WallCombo", 3, token, { HitboxCFrames = {tCF}, BestHitCharacter = tChar, HitCharacters = {tChar}, Ignore = { ActionNumber1 = {tChar} }, DeathInfo = {}, Actions = { ActionNumber1 = {} }, HitInfo = { Blocked = false, IsFacing = true, IsInFront = true }, BlockedCharacters = {}, ServerTime = tick(), FromCFrame = myCF }, "Action71") end)
    pcall(function() Cancel:FireServer(wallCombo) end)
end

function _G.WallCombo:Start()
    if self.loopThread then return end
    self.running = true
    self.loopThread = task.spawn(function()
        while self.running do
            self:UseCombo()
            task.wait(0.02)
        end
        self.loopThread = nil
    end)
end

    local tChar, tHRP = target.Character, target.Character and target.Character:FindFirstChild("HumanoidRootPart")
    local myHRP = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not tHRP or not myHRP then return end

    local tCF, myCF = tHRP.CFrame, myHRP.CFrame

    pcall(function() ReplicatedStorage.Remotes.Replication.ReplicationSpeed:FireServer(0.01) end)
    pcall(function() ReplicatedStorage.Remotes.Abilities.Ability:FireServer(wallCombo, token, nil, tChar, tHRP.Position) end)
    
    local base = { StartCFrameStr = GetCFrameString(tCF), ImpulseVelocity = Vector3.new(-16613, 150000, -65589), AbilityName = "WallCombo", RotVelocityStr = "0,0,0", VelocityStr = "-0.86,2.53,0.04", Gravity = 200000, LookVectorStr = "0,0,0", Duration = 2 }

    local p2 = table.clone(base); p2.RotImpulseVelocity = Vector3.new(-7962, 8808, 7145)
    
    pcall(function() ReplicatedStorage.Remotes.Combat.Action:FireServer(wallCombo, "Characters:"..charName..":WallCombo", 4, token, { HitboxCFrames = {tCF, tCF}, BestHitCharacter = tChar, HitCharacters = {tChar}, Ignore = {}, DeathInfo = {}, Actions = { ActionNumber1 = { [tChar.Name] = p1 } }, HitInfo = { Blocked = false, IsFacing = true, IsInFront = true }, BlockedCharacters = {}, ServerTime = tick(), FromCFrame = myCF }, "Action1254", 0.1) end)

function _G.KillAura:Start()
    if self.loopThread then return end
    self.running = true
    self.fires = 0
    self.loopThread = task.spawn(function()
        while self.running do
            local target = FindNearestPlayerAura()
            if target then
                ExecuteKillAura(target)
                self.fires = self.fires + 1
                task.wait(0.05)
            else task.wait(0.1) end
        end
        self.loopThread = nil
    end)
end

local function StyleButton(btn, text)
    btn.Size = UDim2.new(1, -115, 0, 38)
    btn.BackgroundColor3 = Color3.fromRGB(30, 30, 40)
    btn.BackgroundTransparency = 0.2
    btn.Text = text .. " [OFF]"
    btn.TextColor3 = Color3.fromRGB(200, 200, 200)
    btn.Font = Enum.Font.GothamBold
    btn.TextSize = 12
    btn.BorderSizePixel = 0
    btn.AutoButtonColor = false
    btn:SetAttribute("Active", false)
    
    local pad = Instance.new("UIPadding"); pad.PaddingLeft = UDim.new(0, 12); pad.Parent = btn
    local corner = Instance.new("UICorner"); corner.CornerRadius = UDim.new(0, 6); corner.Parent = btn
    local stroke = Instance.new("UIStroke"); stroke.Color = Color3.fromRGB(60, 60, 80); stroke.Parent = btn
    
    btn.MouseEnter:Connect(function()
        if not btn:GetAttribute("Active") then
            TweenService:Create(btn, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(40, 40, 55), BackgroundTransparency = 0}):Play()
            TweenService:Create(stroke, TweenInfo.new(0.2), {Color = Color3.fromRGB(100, 100, 130)}):Play()
        end
    end)
    btn.MouseLeave:Connect(function()
        if not btn:GetAttribute("Active") then
            TweenService:Create(btn, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(30, 30, 40), BackgroundTransparency = 0.2}):Play()
            TweenService:Create(stroke, TweenInfo.new(0.2), {Color = Color3.fromRGB(60, 60, 80)}):Play()
        end
    end)
    return stroke
end

local draggingCircle, dragStartCircle, startPosCircle = false, nil, nil
CircleIcon.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        draggingCircle = true
        dragStartCircle = input.Position
        startPosCircle = CircleIcon.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then draggingCircle = false end
        end)
    end
end)
CircleIcon.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        if draggingCircle then
            local delta = input.Position - dragStartCircle
            CircleIcon.Position = UDim2.new(startPosCircle.X.Scale, startPosCircle.X.Offset + delta.X, startPosCircle.Y.Scale, startPosCircle.Y.Offset + delta.Y)
        end
    end
end)

local function ToggleStyle(btn, stroke, isOn, color)
    btn:SetAttribute("Active", isOn)
    if isOn then
        TweenService:Create(btn, TweenInfo.new(0.2), {BackgroundColor3 = color, BackgroundTransparency = 0}):Play()
        TweenService:Create(stroke, TweenInfo.new(0.2), {Color = color}):Play()
        btn.TextColor3 = Color3.fromRGB(255, 255, 255)
    else
        TweenService:Create(btn, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(30, 30, 40), BackgroundTransparency = 0.2}):Play()
        TweenService:Create(stroke, TweenInfo.new(0.2), {Color = Color3.fromRGB(60, 60, 80)}):Play()
        btn.TextColor3 = Color3.fromRGB(200, 200, 200)
    end
end
